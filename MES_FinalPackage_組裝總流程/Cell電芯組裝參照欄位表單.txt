use mes;

SELECT
		s.id,
		s.`Time`,
		s.PLCCellID_modinfo,
		t.K_Value,
		t.VAHSB,
		t.VAHSC,	
		s.Class,
		p.PLCTrayID_CE,
		k.acirVP12_CE,
		k.acirRP12_CE,
        s.moduleNO ,
		NULL AS last_define_location,
		NULL AS parallel_match                                    
	FROM mes.stacked_modinfo s
    
	LEFT JOIN mes.testmerge_cc1orcc2 t 
		ON t.modelId = s.PLCCellID_modinfo 
		AND t.parameter = '017'
        
	LEFT JOIN mes.stacked_ce p 
		ON p.PLCCellID_CE = s.PLCCellID_modinfo  
        
	LEFT JOIN mes.schk_cellrule_dupfinish_2 k 
		ON k.PLCCellID_CE = p.PLCCellID_CE

	WHERE
		s.moduleNO = 'M09-14674'
		AND s.`Time` >= DATE_SUB(CURDATE(), INTERVAL 1 DAY)
		AND s.`Time` < DATE_ADD(CURDATE(), INTERVAL 1 DAY)
		AND s.PLCCellID_modinfo IN (
            'MW2016A01204',
			'MW2016A01203',
			'MW2019A08470',
			'MW2019A07204',
			'MW2019A07199',
			'MW2011A34764',
			'MW2019A04063',
			'MW2019A10927',
			'MW2019A00969',
			'MW2019A04992',
			'MW2019A04991',
			'MW2019A04991',
			'MW2019A05614',
			'MW2019A10831',
			'MW2019A04743',
			'MW2019A09258',
			'MW2019A04745',
			'MW2019A04746',
			'MW2019A05463',
			'MW2019A05467',
			'MW2019A05465',
			'MW2019A09904',
			'MW2019A10835',
			'MW2019A10835',
			'MW2019A07416',
			'MW2019A07644',
			'MW2019A02755',
			'MW2019A06979',
			'MW2019A07664',
			'MW2019A07663',
			'MW2019A04628',
			'MW2019A04628'
        ) 
	ORDER BY s.id ASC;
